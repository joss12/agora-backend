
package com.springcommerce.backend.service;

import com.springcommerce.backend.dto.CreateOrderRequest;

import org.junit.jupiter.api.Test;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;

import org.springframework.test.context.ActiveProfiles;

import org.testcontainers.junit.jupiter.Container;
import org.testcontainers.junit.jupiter.Testcontainers;
import org.testcontainers.mysql.MySQLContainer;

import java.math.BigDecimal;

import static org.junit.jupiter.api.Assertions.*;

@SpringBootTest
@Testcontainers
@ActiveProfiles("test")
class OrderServiceIntegrationTest {

    // =========================================================
    // TESTCONTAINERS
    // =========================================================

    @Container
    @ServiceConnection
    static MySQLContainer mysql =
            new MySQLContainer("mysql:8.4");

    // =========================================================
    // SERVICES
    // =========================================================

    @Autowired
    private OrderService orderService;

    @Autowired
    private CartService cartService;

    // =========================================================
    // REPOSITORIES
    // =========================================================

    @Autowired
    private UserRepository userRepository;

    @Autowired
    private ProductRepository productRepository;

    @Autowired
    private OrderRepository orderRepository;

    @Autowired
    private OrderItemRepository orderItemRepository;

    @Autowired
    private CartRepository cartRepository;

    @Autowired
    private CartItemRepository cartItemRepository;

    @Autowired
    private InventoryMovementRepository inventoryMovementRepository;

    @Autowired
    private PaymentRepository paymentRepository;

    @Autowired
    private PaymentAttemptRepository paymentAttemptRepository;

    // =========================================================
    // TEST DATA
    // =========================================================

    private User customer;
    private Product product;

    // =========================================================
    // SETUP
    // =========================================================

    @BeforeEach
    void setUp() {

        // Delete child records before their parent records
        // to avoid foreign-key constraint violations.

        paymentAttemptRepository.deleteAll();
        paymentRepository.deleteAll();

        inventoryMovementRepository.deleteAll();
        orderItemRepository.deleteAll();
        orderRepository.deleteAll();

        cartItemRepository.deleteAll();
        cartRepository.deleteAll();

        productRepository.deleteAll();
        userRepository.deleteAll();

        // Create a test customer.

        customer = new User();

        customer.setFirstName("Test");
        customer.setLastName("Customer");
        customer.setEmail("checkout-test@example.com");
        customer.setPassword("test-hash");
        customer.setRole(UserRole.CUSTOMER);
        customer.setEnabled(true);

        customer = userRepository.save(customer);

        // Create an active product with 10 units in stock.

        product = new Product();

        product.setName("Test Keyboard");
        product.setSku("TEST-KB-001");
        product.setPrice(new BigDecimal("10000.00"));
        product.setStock(10);
        product.setActive(true);

        product = productRepository.save(product);
    }

    // =========================================================
    // DELIVERY ADDRESS HELPER
    // =========================================================

    /**
     * Provides valid Korean delivery information
     * for checkout integration tests.
     */
    private CreateOrderRequest testDeliveryAddress() {

        return new CreateOrderRequest(
                "Kim Minjun",
                "010-1234-5678",
                "06236",
                "123 Teheran-ro, Gangnam-gu, Seoul",
                "Apartment 502",
                "Leave at the door"
        );
    }

    // =========================================================
    // TEST 1: SUCCESSFUL CHECKOUT
    // =========================================================

    /**
     * Successful checkout should:
     *
     * 1. Create an order.
     * 2. Persist the delivery address.
     * 3. Deduct product stock.
     * 4. Create an inventory movement.
     * 5. Create a pending simulated payment.
     * 6. Empty the customer's cart.
     */
    @Test
    void checkoutCreatesOrderAndDeductsStock() {

        cartService.addItem(
                customer.getEmail(),
                new AddCartItemRequest(
                        product.getId(),
                        3
                )
        );

        OrderResponse response = orderService.checkout(
                customer.getEmail(),
                testDeliveryAddress()
        );

        // Verify order details.

        assertNotNull(response.id());

        assertEquals(
                "PLACED",
                response.status()
        );

        assertEquals(
                0,
                new BigDecimal("30000.00")
                        .compareTo(response.totalAmount())
        );

        assertEquals(
                1,
                response.items().size()
        );

        // Verify delivery information in the response.

        assertEquals(
                "Kim Minjun",
                response.recipientName()
        );

        assertEquals(
                "010-1234-5678",
                response.phoneNumber()
        );

        assertEquals(
                "06236",
                response.postalCode()
        );

        assertEquals(
                "123 Teheran-ro, Gangnam-gu, Seoul",
                response.address()
        );

        assertEquals(
                "Apartment 502",
                response.addressDetail()
        );

        assertEquals(
                "Leave at the door",
                response.deliveryInstructions()
        );

        // Verify the delivery information was persisted.

        Order savedOrder = orderRepository
                .findById(response.id())
                .orElseThrow();

        assertEquals(
                "Kim Minjun",
                savedOrder.getRecipientName()
        );

        assertEquals(
                "06236",
                savedOrder.getPostalCode()
        );

        assertEquals(
                "123 Teheran-ro, Gangnam-gu, Seoul",
                savedOrder.getAddress()
        );

        // Verify stock deduction.

        Product updatedProduct = productRepository
                .findById(product.getId())
                .orElseThrow();

        assertEquals(
                7,
                updatedProduct.getStock()
        );

        // Verify the order and order item.

        assertEquals(
                1,
                orderRepository.count()
        );

        assertEquals(
                1,
                orderItemRepository.count()
        );

        // Verify the inventory movement.

        assertEquals(
                1,
                inventoryMovementRepository.count()
        );

        // Verify the simulated payment.

        assertEquals(
                1,
                paymentRepository.count()
        );

        Payment payment = paymentRepository
                .findByOrderId(response.id())
                .orElseThrow();

        assertEquals(
                PaymentStatus.PENDING,
                payment.getStatus()
        );

        assertEquals(
                0,
                new BigDecimal("30000.00")
                        .compareTo(payment.getAmount())
        );

        assertNull(
                payment.getTransactionReference()
        );

        // No payment has been simulated yet.

        assertEquals(
                0,
                paymentAttemptRepository.count()
        );

        // Verify the cart is empty.

        Long cartId = cartRepository
                .findByUserId(customer.getId())
                .orElseThrow()
                .getId();

        assertTrue(
                cartItemRepository
                        .findByCartIdOrderByCreatedAtAsc(cartId)
                        .isEmpty()
        );
    }

    // =========================================================
    // TEST 2: INSUFFICIENT STOCK
    // =========================================================

    /**
     * Checkout must roll back completely when the
     * requested quantity exceeds available stock.
     */
    @Test
    void insufficientStockRollsBackCheckout() {

        cartService.addItem(
                customer.getEmail(),
                new AddCartItemRequest(
                        product.getId(),
                        20
                )
        );

        assertThrows(
                InsufficientStockException.class,
                () -> orderService.checkout(
                        customer.getEmail(),
                        testDeliveryAddress()
                )
        );

        // No order or order items should exist.

        assertEquals(
                0,
                orderRepository.count()
        );

        assertEquals(
                0,
                orderItemRepository.count()
        );

        // No simulated payment should exist.

        assertEquals(
                0,
                paymentRepository.count()
        );

        assertEquals(
                0,
                paymentAttemptRepository.count()
        );

        // Stock must remain unchanged.

        assertEquals(
                10,
                productRepository
                        .findById(product.getId())
                        .orElseThrow()
                        .getStock()
        );

        // No inventory movement should exist.

        assertEquals(
                0,
                inventoryMovementRepository.count()
        );

        // The customer's cart must remain intact.

        Long cartId = cartRepository
                .findByUserId(customer.getId())
                .orElseThrow()
                .getId();

        assertEquals(
                1,
                cartItemRepository
                        .findByCartIdOrderByCreatedAtAsc(cartId)
                        .size()
        );
    }

    // =========================================================
    // TEST 3: ARCHIVED PRODUCT
    // =========================================================

    /**
     * A product archived after being added to the cart
     * must not be purchasable.
     */
    @Test
    void archivedProductCannotBePurchased() {

        // Add the product while it is active.

        cartService.addItem(
                customer.getEmail(),
                new AddCartItemRequest(
                        product.getId(),
                        2
                )
        );

        // Archive the product before checkout.

        product.setActive(false);

        productRepository.saveAndFlush(
                product
        );

        assertThrows(
                ProductNotFoundException.class,
                () -> orderService.checkout(
                        customer.getEmail(),
                        testDeliveryAddress()
                )
        );

        // No order or order items should exist.

        assertEquals(
                0,
                orderRepository.count()
        );

        assertEquals(
                0,
                orderItemRepository.count()
        );

        // No simulated payment should exist.

        assertEquals(
                0,
                paymentRepository.count()
        );

        assertEquals(
                0,
                paymentAttemptRepository.count()
        );

        // Stock must remain unchanged.

        assertEquals(
                10,
                productRepository
                        .findById(product.getId())
                        .orElseThrow()
                        .getStock()
        );

        // No inventory movement should exist.

        assertEquals(
                0,
                inventoryMovementRepository.count()
        );

        // The original cart item must remain.

        Long cartId = cartRepository
                .findByUserId(customer.getId())
                .orElseThrow()
                .getId();

        assertEquals(
                1,
                cartItemRepository
                        .findByCartIdOrderByCreatedAtAsc(cartId)
                        .size()
        );
    }

    // =========================================================
    // TEST 4: CANCEL UNPAID ORDER
    // =========================================================

    /**
     * Cancelling a pending-payment order should restore
     * stock and mark the order as cancelled.
     */
    @Test
    void cancellingUnpaidOrderRestoresStock() {

        cartService.addItem(
                customer.getEmail(),
                new AddCartItemRequest(
                        product.getId(),
                        3
                )
        );

        OrderResponse placedOrder = orderService.checkout(
                customer.getEmail(),
                testDeliveryAddress()
        );

        OrderResponse cancelledOrder = orderService.cancelOrder(
                customer.getEmail(),
                placedOrder.id()
        );

        assertEquals(
                "CANCELLED",
                cancelledOrder.status()
        );

        // Stock should return to its original quantity.

        assertEquals(
                10,
                productRepository
                        .findById(product.getId())
                        .orElseThrow()
                        .getStock()
        );

        // One STOCK_OUT and one STOCK_IN movement.

        assertEquals(
                2,
                inventoryMovementRepository.count()
        );

        // Order history must be preserved.

        assertEquals(
                1,
                orderRepository.count()
        );

        assertEquals(
                1,
                orderItemRepository.count()
        );

        // An unpaid payment remains PENDING.
        // No refund is necessary.

        Payment payment = paymentRepository
                .findByOrderId(placedOrder.id())
                .orElseThrow();

        assertEquals(
                PaymentStatus.PENDING,
                payment.getStatus()
        );

        assertEquals(
                0,
                paymentAttemptRepository.count()
        );
    }
}

@SpringBootTest
@Testcontainers
@ActiveProfiles("test")
class AdminOrderServiceIntegrationTest {

    @Container
    @ServiceConnection
    static MySQLContainer mysql =
            new MySQLContainer("mysql:8.4");

    // =========================================================
    // SERVICES
    // =========================================================

    @Autowired
    private AdminOrderService adminOrderService;

    @Autowired
    private OrderService orderService;

    @Autowired
    private PaymentService paymentService;

    @Autowired
    private ShipmentService shipmentService;

    @Autowired
    private CartService cartService;

    // =========================================================
    // REPOSITORIES
    // =========================================================

    @Autowired
    private UserRepository userRepository;

    @Autowired
    private ProductRepository productRepository;

    @Autowired
    private CartRepository cartRepository;

    @Autowired
    private CartItemRepository cartItemRepository;

    @Autowired
    private OrderRepository orderRepository;

    @Autowired
    private OrderItemRepository orderItemRepository;

    @Autowired
    private InventoryMovementRepository inventoryMovementRepository;

    @Autowired
    private PaymentRepository paymentRepository;

    @Autowired
    private PaymentAttemptRepository paymentAttemptRepository;

    @Autowired
    private ShipmentRepository shipmentRepository;

    // =========================================================
    // TEST DATA
    // =========================================================

    private User customer;
    private Product product;

    @BeforeEach
    void setUp() {

        // Delete dependent records first to avoid
        // foreign-key constraint violations.

        shipmentRepository.deleteAll();

        paymentAttemptRepository.deleteAll();
        paymentRepository.deleteAll();

        inventoryMovementRepository.deleteAll();
        orderItemRepository.deleteAll();
        orderRepository.deleteAll();

        cartItemRepository.deleteAll();
        cartRepository.deleteAll();

        productRepository.deleteAll();
        userRepository.deleteAll();

        // -----------------------------------------------------

        // -----------------------------------------------------

        customer = new User();

        customer.setFirstName("Test");
        customer.setLastName("Customer");

        // -----------------------------------------------------

        product = new Product();

        product.setSku("ADMIN-MONITOR-001");

    // =========================================================
    // HELPERS
    // =========================================================

    private CreateOrderRequest deliveryAddress() {

        return new CreateOrderRequest(
                "Kim Minjun",
                "010-1234-5678",
                "06236",
                "123 Teheran-ro, Gangnam-gu, Seoul",
                "Apartment 502",
                "Leave at the door"
        );
    }

    private OrderResponse placeOrder() {

        cartService.addItem(
                customer.getEmail(),
                new AddCartItemRequest(
                        product.getId(),
                        2
                )
        );

        return orderService.checkout(
                customer.getEmail(),
                deliveryAddress()
        );
    }

    private OrderResponse placeAndPayOrder() {

        OrderResponse order = placeOrder();

        paymentService.simulate(
                customer.getEmail(),
                order.id(),
                PaymentOutcome.SUCCESS
        );

        return order;
    }

    private void assertOrderStatus(
            Long orderId,
            OrderStatus expectedStatus
    ) {

        OrderStatus actualStatus = orderRepository
                .findById(orderId)
                .orElseThrow()
                .getStatus();

        assertEquals(expectedStatus, actualStatus);
    }

    private void assertProductStock(int expectedStock) {

        int actualStock = productRepository
                .findById(product.getId())
                .orElseThrow()
                .getStock();

        assertEquals(expectedStock, actualStock);
    }

    // =========================================================
    // TEST 1: ADMIN ORDER LIST
    // =========================================================

        OrderResponse order = placeOrder();

        Page<AdminOrderResponse> page =
                adminOrderService.getOrders(

                        PageRequest.of(0, 10)
                );

        assertEquals(1, page.getTotalElements());

        AdminOrderResponse result =
                page.getContent().get(0);

        assertEquals(order.id(), result.id());
        assertEquals(customer.getId(), result.customerId());
        assertEquals(customer.getEmail(), result.customerEmail());
        assertEquals("PLACED", result.status());
    }

                );

        Page<AdminOrderResponse> shipped =
                adminOrderService.getOrders(
                        OrderStatus.SHIPPED,

        assertEquals(1, placed.getTotalElements());

    }

    // =========================================================
    // TEST 3: ADMIN ORDER DETAILS
    // =========================================================

    @Test
    void adminCanViewOrderDetails() {

        AdminOrderResponse response =

        assertEquals(order.id(), response.id());

    // =========================================================
    // TEST 4: UNPAID ORDER
    // =========================================================

    @Test
    void unpaidOrderCannotBeProcessed() {

        OrderResponse order = placeOrder();

        assertThrows(
                OrderStatusConflictException.class,
                () -> adminOrderService.updateStatus(
                        order.id(),
                        OrderStatus.PROCESSING
                )
        );

        assertOrderStatus(
                order.id(),
                OrderStatus.PLACED
        );

        assertProductStock(8);
    }

    // =========================================================
    // TEST 5: FULL FULFILLMENT
    // =========================================================

    @Test
    void paidOrderCanProgressThroughFulfillment() {

        // -----------------------------------------------------
        // 1. CUSTOMER PLACES AND PAYS FOR ORDER
        // -----------------------------------------------------

        OrderResponse order = placeAndPayOrder();

        Payment payment = paymentRepository
                .findByOrderId(order.id())
                .orElseThrow();

        assertEquals(
                PaymentStatus.PAID,
                payment.getStatus()
        );

        assertOrderStatus(
                order.id(),
                OrderStatus.PLACED
        );

        assertProductStock(8);

        // -----------------------------------------------------
        // 2. ADMIN STARTS PROCESSING
        // -----------------------------------------------------

        AdminOrderResponse processing =
                adminOrderService.updateStatus(
                        order.id(),
                        OrderStatus.PROCESSING
                );

        assertEquals(
                "PROCESSING",
                processing.status()
        );

        assertOrderStatus(
                order.id(),
                OrderStatus.PROCESSING
        );

        // -----------------------------------------------------
        // 3. ADMIN CREATES SHIPMENT
        // -----------------------------------------------------

        shipmentService.shipOrder(
                order.id(),
                new CreateShipmentRequest(
                        "CJ Logistics",
                        "AGORA-TRACK-001"
                )
        );

        assertOrderStatus(
                order.id(),
                OrderStatus.SHIPPED
        );

        Shipment shipment = shipmentRepository
                .findByOrderId(order.id())
                .orElseThrow();

        assertEquals(
                ShipmentStatus.SHIPPED,
                shipment.getStatus()
        );

        assertEquals(
                "CJ Logistics",
                shipment.getCarrier()
        );

        assertEquals(
                "AGORA-TRACK-001",
                shipment.getTrackingNumber()
        );

        assertNotNull(shipment.getShippedAt());

        // -----------------------------------------------------
        // 4. CUSTOMER CONFIRMS DELIVERY
        // -----------------------------------------------------

        shipmentService.confirmCustomerDelivery(
                customer.getEmail(),
                order.id()
        );

        assertOrderStatus(
                order.id(),
                OrderStatus.DELIVERED
        );

        Shipment deliveredShipment = shipmentRepository
                .findByOrderId(order.id())
                .orElseThrow();

        assertEquals(
                ShipmentStatus.DELIVERED,
                deliveredShipment.getStatus()
        );

        assertNotNull(
                deliveredShipment.getDeliveredAt()
        );

        // -----------------------------------------------------
        // 5. INVENTORY MUST NOT BE DEDUCTED AGAIN
        // -----------------------------------------------------

        assertProductStock(8);

        // -----------------------------------------------------
        // 6. PAYMENT MUST REMAIN PAID
        // -----------------------------------------------------

        Payment completedPayment = paymentRepository
                .findByOrderId(order.id())
                .orElseThrow();

        assertEquals(
                PaymentStatus.PAID,
                completedPayment.getStatus()
        );
    }

    // =========================================================
    // TEST 6: INVALID STATUS TRANSITION
    // =========================================================

    @Test
    void adminCannotSkipFulfillmentStages() {

        OrderResponse order = placeAndPayOrder();

        // Admin cannot ship directly through updateStatus.

        assertThrows(
                OrderStatusConflictException.class,
                () -> adminOrderService.updateStatus(
                        order.id(),
                        OrderStatus.SHIPPED
                )
        );

        // Admin cannot mark an order delivered directly.

        assertThrows(
                OrderStatusConflictException.class,
                () -> adminOrderService.updateStatus(
                        order.id(),
                        OrderStatus.DELIVERED
                )
        );

        assertOrderStatus(
                order.id(),
                OrderStatus.PLACED
        );

        assertProductStock(8);
    }

    // =========================================================
    // TEST 7: ADMIN CANCELS UNPAID ORDER
    // =========================================================

    @Test
    void adminCanCancelUnpaidOrder() {

        OrderResponse order = placeOrder();

        assertProductStock(8);

        AdminOrderResponse cancelled =
                adminOrderService.updateStatus(
                        order.id(),
                        OrderStatus.CANCELLED
                );

        assertEquals(
                "CANCELLED",
                cancelled.status()
        );

        assertOrderStatus(
                order.id(),
                OrderStatus.CANCELLED
        );

        // Inventory must be restored.

        assertProductStock(10);

        // No successful payment occurred.

        Payment payment = paymentRepository
                .findByOrderId(order.id())

        assertEquals(
                PaymentStatus.PENDING,
                payment.getStatus()
        );
    }

    // =========================================================
    // TEST 8: ADMIN CANCELS PAID ORDER

    void adminCanCancelPaidOrder() {

                adminOrderService.updateStatus(

    // =========================================================
    // TEST 12: CANNOT PROCESS SAME ORDER TWICE
    // =========================================================

    @Test
    void adminCannotProcessOrderTwice() {

        OrderResponse order = placeAndPayOrder();

        adminOrderService.updateStatus(
                order.id(),
                OrderStatus.PROCESSING
        );

        assertThrows(
                OrderStatusConflictException.class,
                () -> adminOrderService.updateStatus(
                        order.id(),
                        OrderStatus.PROCESSING
                )
        );

        assertOrderStatus(
                order.id(),
                OrderStatus.PROCESSING
        );

        assertProductStock(8);
    }

    // =========================================================
    // TEST 13: FULFILLMENT PRESERVES ORDER TOTAL
    // =========================================================

    @Test
    void fulfillmentPreservesOrderTotal() {

        OrderResponse order = placeAndPayOrder();

        BigDecimal originalTotal = order.totalAmount();

        adminOrderService.updateStatus(
                order.id(),
                OrderStatus.PROCESSING
        );

        shipmentService.shipOrder(
                order.id(),
                new CreateShipmentRequest(
                        "CJ Logistics",
                        "AGORA-TRACK-003"
                )
        );

        shipmentService.markDelivered(order.id());

        AdminOrderResponse completedOrder =
                adminOrderService.getOrder(order.id());

        assertEquals(
                0,
                originalTotal.compareTo(

                )
        );

                "DELIVERED",
                completedOrder.status()
        );

        assertProductStock(8);
    }

@SpringBootTest
@Testcontainers
@ActiveProfiles("test")
class PaymentServiceIntegrationTest {

    @Container
    @ServiceConnection
    static MySQLContainer mysql =
            new MySQLContainer("mysql:8.4");

    @Autowired
    private OrderService orderService;

    @Autowired
    private PaymentService paymentService;

    @Autowired
    private CartService cartService;

    @Autowired
    private UserRepository userRepository;

    @Autowired
    private ProductRepository productRepository;

    @Autowired
    private CartRepository cartRepository;

    @Autowired
    private CartItemRepository cartItemRepository;

    @Autowired
    private OrderRepository orderRepository;

    @Autowired
    private OrderItemRepository orderItemRepository;

    @Autowired
    private InventoryMovementRepository inventoryMovementRepository;

    @Autowired
    private PaymentRepository paymentRepository;

    @Autowired
    private PaymentAttemptRepository paymentAttemptRepository;

    private User customer;
    private Product product;

    @BeforeEach
    void setUp() {

        paymentAttemptRepository.deleteAll();
        paymentRepository.deleteAll();

        inventoryMovementRepository.deleteAll();
        orderItemRepository.deleteAll();
        orderRepository.deleteAll();

        cartItemRepository.deleteAll();
        cartRepository.deleteAll();

        productRepository.deleteAll();
        userRepository.deleteAll();

        customer = new User();
        customer.setFirstName("Test");
        customer.setLastName("Customer");
        customer.setEmail("payment-test@example.com");
        customer.setPassword("test-hash");
        customer.setRole(UserRole.CUSTOMER);
        customer.setEnabled(true);

        customer = userRepository.save(customer);

        product = new Product();
        product.setName("Test Keyboard");
        product.setSku("PAYMENT-KB-001");
        product.setPrice(new BigDecimal("10000.00"));
        product.setStock(10);
        product.setActive(true);

        product = productRepository.save(product);
    }

    private CreateOrderRequest deliveryAddress() {

        return new CreateOrderRequest(
                "Kim Minjun",
                "010-1234-5678",
                "06236",
                "123 Teheran-ro, Gangnam-gu, Seoul",
                "Apartment 502",
                "Leave at the door"
        );
    }

    private OrderResponse placeOrder() {

        cartService.addItem(
                customer.getEmail(),
                new AddCartItemRequest(
                        product.getId(),
                        2
                )
        );

        return orderService.checkout(
                customer.getEmail(),
                deliveryAddress()
        );
    }

    // =========================================================
    // TEST 1: SUCCESSFUL PAYMENT
    // =========================================================

    @Test
    void successfulPaymentMarksPaymentAsPaid() {

        OrderResponse order = placeOrder();

        PaymentResponse response = paymentService.simulate(
                customer.getEmail(),
                order.id(),
                PaymentOutcome.SUCCESS
        );

        assertEquals("PAID", response.status());

        assertNotNull(response.transactionReference());

        assertEquals(1, response.attempts().size());

        assertEquals(
                1,
                paymentAttemptRepository.count()
        );

        Payment payment = paymentRepository
                .findByOrderId(order.id())
                .orElseThrow();

        assertEquals(
                PaymentStatus.PAID,
                payment.getStatus()
        );

        assertEquals(
                0,
                new BigDecimal("20000.00")
                        .compareTo(payment.getAmount())
        );

        // Paying does not advance fulfillment automatically.
        assertEquals(
                OrderStatus.PLACED,
                orderRepository.findById(order.id())
                        .orElseThrow()
                        .getStatus()
        );
    }

    // =========================================================
    // TEST 2: FAILED PAYMENT
    // =========================================================

    @Test
    void failedPaymentRecordsAttemptAndAllowsRetry() {

        OrderResponse order = placeOrder();

        PaymentResponse response = paymentService.simulate(
                customer.getEmail(),
                order.id(),
                PaymentOutcome.FAILURE
        );

        assertEquals("FAILED", response.status());

        assertEquals(
                1,
                response.attempts().size()
        );

        assertEquals(
                "FAILURE",
                response.attempts().get(0).outcome()
        );

        assertEquals(
                PaymentStatus.FAILED,
                paymentRepository.findByOrderId(order.id())
                        .orElseThrow()
                        .getStatus()
        );

        // A declined payment must not restore stock.
        assertEquals(
                8,
                productRepository.findById(product.getId())
                        .orElseThrow()
                        .getStock()
        );
    }

    // =========================================================
    // TEST 3: RETRY FAILED PAYMENT
    // =========================================================

    @Test
    void failedPaymentCanBeRetriedSuccessfully() {

        OrderResponse order = placeOrder();

        paymentService.simulate(
                customer.getEmail(),
                order.id(),
                PaymentOutcome.FAILURE
        );

        PaymentResponse response = paymentService.simulate(
                customer.getEmail(),
                order.id(),
                PaymentOutcome.SUCCESS
        );

        assertEquals("PAID", response.status());

        assertEquals(
                2,
                response.attempts().size()
        );

        assertEquals(
                "FAILURE",
                response.attempts().get(0).outcome()
        );

        assertEquals(
                "SUCCESS",
                response.attempts().get(1).outcome()
        );

        assertEquals(
                2,
                paymentAttemptRepository.count()
        );

        // Retrying payment must not deduct stock again.
        assertEquals(
                8,
                productRepository.findById(product.getId())
                        .orElseThrow()
                        .getStock()
        );
    }

    // =========================================================
    // TEST 4: DUPLICATE SUCCESSFUL PAYMENT
    // =========================================================

    @Test
    void paidOrderCannotBePaidAgain() {

        OrderResponse order = placeOrder();

        paymentService.simulate(
                customer.getEmail(),
                order.id(),
                PaymentOutcome.SUCCESS
        );

        assertThrows(
                PaymentConflictException.class,
                () -> paymentService.simulate(
                        customer.getEmail(),
                        order.id(),
                        PaymentOutcome.SUCCESS
                )
        );

        assertEquals(
                1,
                paymentAttemptRepository.count()
        );
    }

    // =========================================================
    // TEST 5: REFUND ON CANCELLATION
    // =========================================================

    @Test
    void cancellingPaidOrderSimulatesRefund() {

        OrderResponse order = placeOrder();

        paymentService.simulate(
                customer.getEmail(),
                order.id(),
                PaymentOutcome.SUCCESS
        );

        OrderResponse cancelled = orderService.cancelOrder(
                customer.getEmail(),
                order.id()
        );

        assertEquals(
                "CANCELLED",
                cancelled.status()
        );

        Payment payment = paymentRepository
                .findByOrderId(order.id())
                .orElseThrow();

        assertEquals(
                PaymentStatus.REFUNDED,
                payment.getStatus()
        );

        // Inventory must be restored.
        assertEquals(
                10,
                productRepository.findById(product.getId())
                        .orElseThrow()
                        .getStock()
        );

        // The original payment attempt is retained.
        assertEquals(
                1,
                paymentAttemptRepository.count()
        );
    }

    // =========================================================
    // TEST 6: ORDER OWNERSHIP
    // =========================================================

    @Test
    void anotherCustomerCannotAccessPayment() {

        OrderResponse order = placeOrder();

        User otherCustomer = new User();
        otherCustomer.setFirstName("Other");
        otherCustomer.setLastName("Customer");
        otherCustomer.setEmail("other-payment@example.com");
        otherCustomer.setPassword("test-hash");
        otherCustomer.setRole(UserRole.CUSTOMER);
        otherCustomer.setEnabled(true);

        userRepository.save(otherCustomer);

        assertThrows(
                OrderNotFoundException.class,
                () -> paymentService.getPayment(
                        otherCustomer.getEmail(),
                        order.id()
                )
        );

        assertThrows(
                OrderNotFoundException.class,
                () -> paymentService.simulate(
                        otherCustomer.getEmail(),
                        order.id(),
                        PaymentOutcome.SUCCESS
                )
        );

        assertEquals(
                PaymentStatus.PENDING,
                paymentRepository.findByOrderId(order.id())
                        .orElseThrow()
                        .getStatus()
        );

        assertEquals(
                0,
                paymentAttemptRepository.count()
        );
    }
