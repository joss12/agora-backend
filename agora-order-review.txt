@Service
public class OrderService {

    private final UserRepository userRepository;
    private final CartRepository cartRepository;
    private final CartItemRepository cartItemRepository;
    private final ProductRepository productRepository;
    private final OrderRepository orderRepository;
    private final OrderItemRepository orderItemRepository;
    private final InventoryMovementRepository inventoryMovementRepository;
    private final PaymentRepository paymentRepository;

    public OrderService(
            UserRepository userRepository,
            CartRepository cartRepository,
            CartItemRepository cartItemRepository,
            ProductRepository productRepository,
            OrderRepository orderRepository,
            OrderItemRepository orderItemRepository,
            InventoryMovementRepository inventoryMovementRepository,
            PaymentRepository paymentRepository
    ) {
        this.userRepository = userRepository;
        this.cartRepository = cartRepository;
        this.cartItemRepository = cartItemRepository;
        this.productRepository = productRepository;
        this.orderRepository = orderRepository;
        this.orderItemRepository = orderItemRepository;
        this.inventoryMovementRepository = inventoryMovementRepository;
        this.paymentRepository = paymentRepository;
    }

    // =========================================================
    // CHECKOUT
    // =========================================================

    @Transactional
    @CacheEvict(
            cacheNames = "products",
            allEntries = true
    )
    public OrderResponse checkout(
            String email,
            CreateOrderRequest request
    ) {

        User user = findUser(email);

        /*
         * Lock the customer's cart before reading its items.
         *
         * Another checkout using the same lock must wait
         * until this transaction commits or rolls back.
         */
        Cart cart = cartRepository
                .findByUserIdForUpdate(user.getId())
                .orElseThrow(EmptyCartException::new);

        List<CartItem> cartItems = new ArrayList<>(
                cartItemRepository
                        .findByCartIdOrderByCreatedAtAsc(
                                cart.getId()
                        )
        );

        if (cartItems.isEmpty()) {
            throw new EmptyCartException();
        }

        /*
         * Acquire product locks in ascending ID order.
         * Consistent ordering reduces the risk of deadlocks.
         */
        cartItems.sort(
                Comparator.comparing(
                        item -> item.getProduct().getId()
                )
        );

        // -----------------------------------------------------
        // CREATE ORDER
        // -----------------------------------------------------

        Order order = new Order();

        order.setUser(user);
        order.setStatus(OrderStatus.PLACED);
        order.setTotalAmount(BigDecimal.ZERO);

        order.setRecipientName(
                request.recipientName().trim()
        );

        order.setPhoneNumber(
                request.phoneNumber().trim()
        );

        order.setPostalCode(
                request.postalCode().trim()
        );

        order.setAddress(
                request.address().trim()
        );

        order.setAddressDetail(
                request.addressDetail() == null
                        ? null
                        : request.addressDetail().trim()
        );

        order.setDeliveryInstructions(
                request.deliveryInstructions() == null
                        ? null
                        : request.deliveryInstructions().trim()
        );

        Order savedOrder = orderRepository.save(order);

        List<OrderItem> orderItems = new ArrayList<>();

        BigDecimal orderTotal = BigDecimal.ZERO;

        // -----------------------------------------------------
        // RESERVE INVENTORY
        // -----------------------------------------------------

        for (CartItem cartItem : cartItems) {

            Long productId = cartItem
                    .getProduct()
                    .getId();

            Product product = productRepository
                    .findByIdForUpdate(productId)
                    .orElseThrow(
                            () -> new ProductNotFoundException(
                                    productId
                            )
                    );

            if (!product.isActive()) {
                throw new ProductNotFoundException(
                        productId
                );
            }

            int quantity = cartItem.getQuantity();

            if (quantity <= 0) {
                throw new IllegalStateException(
                        "Cart item quantity must be greater than zero"
                );
            }

            int stockBefore = product.getStock();

            if (stockBefore < quantity) {
                throw new InsufficientStockException(
                        product.getId(),
                        stockBefore,
                        quantity
                );
            }

            int stockAfter = stockBefore - quantity;

            BigDecimal subtotal = product
                    .getPrice()
                    .multiply(
                            BigDecimal.valueOf(quantity)
                    );

            // Save a snapshot of the purchased product.

            OrderItem orderItem = new OrderItem();

            orderItem.setOrder(savedOrder);
            orderItem.setProduct(product);
            orderItem.setProductName(product.getName());
            orderItem.setSku(product.getSku());
            orderItem.setUnitPrice(product.getPrice());
            orderItem.setQuantity(quantity);
            orderItem.setSubtotal(subtotal);

            orderItems.add(orderItem);

            orderTotal = orderTotal.add(subtotal);

            // Deduct stock.

            product.setStock(stockAfter);

            productRepository.save(product);

            // Record the stock movement.

            InventoryMovement movement =
                    new InventoryMovement();

            movement.setProduct(product);
            movement.setType(
                    InventoryMovementType.STOCK_OUT
            );
            movement.setQuantity(quantity);
            movement.setStockBefore(stockBefore);
            movement.setStockAfter(stockAfter);

            inventoryMovementRepository.save(movement);
        }

        // -----------------------------------------------------
        // SAVE ORDER ITEMS AND TOTAL
        // -----------------------------------------------------

        orderItemRepository.saveAll(orderItems);

        savedOrder.setTotalAmount(orderTotal);

        orderRepository.save(savedOrder);

        // -----------------------------------------------------
        // CREATE PENDING PAYMENT
        // -----------------------------------------------------

        Payment payment = new Payment();

        payment.setOrder(savedOrder);
        payment.setAmount(orderTotal);
        payment.setStatus(PaymentStatus.PENDING);

        paymentRepository.save(payment);

        // -----------------------------------------------------
        // EMPTY CART
        // -----------------------------------------------------

        cartItemRepository.deleteByCartId(
                cart.getId()
        );

        return mapToResponse(
                savedOrder,
                orderItems
        );
    }

    // =========================================================
    // CUSTOMER ORDER HISTORY
    // =========================================================

    @Transactional(readOnly = true)
    public List<OrderResponse> getOrders(
            String email
    ) {

        User user = findUser(email);

        return orderRepository
                .findByUserIdOrderByCreatedAtDesc(
                        user.getId()
                )
                .stream()
                .map(
                        order -> mapToResponse(
                                order,
                                orderItemRepository
                                        .findByOrderIdOrderByIdAsc(
                                                order.getId()
                                        )
                        )
                )
                .toList();
    }

    // =========================================================
    // CUSTOMER ORDER DETAILS
    // =========================================================

    @Transactional(readOnly = true)
    public OrderResponse getOrder(
            String email,
            Long orderId
    ) {

        User user = findUser(email);

        Order order = orderRepository
                .findByIdAndUserId(
                        orderId,
                        user.getId()
                )
                .orElseThrow(
                        () -> new OrderNotFoundException(
                                orderId
                        )
                );

        List<OrderItem> orderItems =
                orderItemRepository
                        .findByOrderIdOrderByIdAsc(
                                order.getId()
                        );

        return mapToResponse(
                order,
                orderItems
        );
    }

    // =========================================================
    // CANCEL ORDER
    // =========================================================

    @Transactional
    @CacheEvict(
            cacheNames = "products",
            allEntries = true
    )
    public OrderResponse cancelOrder(
            String email,
            Long orderId
    ) {

        User user = findUser(email);

        /*
         * Lock the order so cancellation and payment
         * simulation cannot update it simultaneously
         * without coordinating on the same row.
         */
        Order order = orderRepository
                .findByIdAndUserIdForUpdate(
                        orderId,
                        user.getId()
                )
                .orElseThrow(
                        () -> new OrderNotFoundException(
                                orderId
                        )
                );

        if (order.getStatus() != OrderStatus.PLACED) {
            throw new OrderCancellationException(
                    orderId
            );
        }

        Payment payment = paymentRepository
                .findByOrderId(orderId)
                .orElseThrow(
                        () -> new PaymentNotFoundException(
                                orderId
                        )
                );

        List<OrderItem> orderItems =
                new ArrayList<>(
                        orderItemRepository
                                .findByOrderIdOrderByIdAsc(
                                        orderId
                                )
                );

        /*
         * Restore product stock in a consistent
         * locking order.
         */
        orderItems.sort(
                Comparator.comparing(
                        item -> item.getProduct().getId()
                )
        );

        for (OrderItem item : orderItems) {

            Long productId = item
                    .getProduct()
                    .getId();

            Product product = productRepository
                    .findByIdForUpdate(productId)
                    .orElseThrow(
                            () -> new ProductNotFoundException(
                                    productId
                            )
                    );

            int quantity = item.getQuantity();

            int stockBefore = product.getStock();

            int stockAfter = Math.addExact(
                    stockBefore,
                    quantity
            );

            product.setStock(stockAfter);

            productRepository.save(product);

            InventoryMovement movement =
                    new InventoryMovement();

            movement.setProduct(product);
            movement.setType(
                    InventoryMovementType.STOCK_IN
            );
            movement.setQuantity(quantity);
            movement.setStockBefore(stockBefore);
            movement.setStockAfter(stockAfter);

            inventoryMovementRepository.save(movement);
        }

        // -----------------------------------------------------
        // SIMULATED REFUND
        // -----------------------------------------------------

        if (payment.getStatus() == PaymentStatus.PAID) {

            payment.setStatus(
                    PaymentStatus.REFUNDED
            );

            paymentRepository.save(payment);

        } else if (
                payment.getStatus()
                        == PaymentStatus.REFUNDED
        ) {

            throw new IllegalStateException(
                    "Payment has already been refunded"
            );
        }

        /*
         * PENDING and FAILED payments do not require
         * a refund because no successful payment occurred.
         */

        order.setStatus(
                OrderStatus.CANCELLED
        );

        orderRepository.save(order);

        return mapToResponse(
                order,
                orderItems
        );
    }

    // =========================================================
    // HELPERS
    // =========================================================

    private User findUser(
            String email
    ) {

        String normalizedEmail = email
                .trim()
                .toLowerCase(Locale.ROOT);

        return userRepository
                .findByEmail(normalizedEmail)
                .orElseThrow(
                        () -> new IllegalStateException(
                                "Authenticated user not found"
                        )
                );
    }

    private OrderResponse mapToResponse(
            Order order,
            List<OrderItem> orderItems
    ) {

        List<OrderItemResponse> items =
                orderItems
                        .stream()
                        .map(
                                item -> new OrderItemResponse(
                                        item.getId(),
                                        item.getProduct().getId(),
                                        item.getProductName(),
                                        item.getSku(),
                                        item.getUnitPrice(),
                                        item.getQuantity(),
                                        item.getSubtotal()
                                )
                        )
                        .toList();

        return new OrderResponse(
                order.getId(),
                order.getStatus().name(),
                order.getTotalAmount(),
                items,
                order.getRecipientName(),
                order.getPhoneNumber(),
                order.getPostalCode(),
                order.getAddress(),
                order.getAddressDetail(),
                order.getDeliveryInstructions(),
                order.getCreatedAt()
        );
    }
}
package com.springcommerce.backend.service;
import org.springframework.cache.annotation.CacheEvict;

import com.springcommerce.backend.dto.AdminOrderResponse;

import java.util.ArrayList;

@Service
public class AdminOrderService {

    private final OrderRepository orderRepository;
    private final OrderItemRepository orderItemRepository;
    private final PaymentRepository paymentRepository;
    private final ProductRepository productRepository;
    private final InventoryMovementRepository inventoryMovementRepository;

    public AdminOrderService(
            OrderRepository orderRepository,
            OrderItemRepository orderItemRepository,
            PaymentRepository paymentRepository,
            ProductRepository productRepository,
            InventoryMovementRepository inventoryMovementRepository
    ) {
        this.orderRepository = orderRepository;
        this.orderItemRepository = orderItemRepository;
        this.paymentRepository = paymentRepository;
        this.productRepository = productRepository;
        this.inventoryMovementRepository = inventoryMovementRepository;
    }

    @Transactional(readOnly = true)
    public Page<AdminOrderResponse> getOrders(
            OrderStatus status,
            Pageable pageable
    ) {
        Page<Order> orders = status == null
                ? orderRepository.findAllByOrderByCreatedAtDesc(pageable)
                : orderRepository.findByStatusOrderByCreatedAtDesc(
                status,
                pageable
        );

        return orders.map(this::mapToResponse);
    }

    @Transactional(readOnly = true)
    public AdminOrderResponse getOrder(
            Long orderId
    ) {
        Order order = orderRepository
                .findById(orderId)
                .orElseThrow(
                        () -> new OrderNotFoundException(orderId)
                );

        return mapToResponse(order);
    }

    @Transactional(readOnly = true)
    public List<AdminOrderResponse> getOrdersByCustomer(
            Long customerId
    ) {
        return orderRepository
                .findByUserIdOrderByCreatedAtDesc(customerId)
                .stream()
                .map(this::mapToResponse)
                .toList();
    }

    @Transactional
    @CacheEvict(
            cacheNames = "products",
            allEntries = true
    )
    public AdminOrderResponse updateStatus(
            Long orderId,
            OrderStatus requestedStatus
    ) {
        Order order = orderRepository
                .findByIdForUpdate(orderId)
                .orElseThrow(
                        () -> new OrderNotFoundException(orderId)
                );

        if (requestedStatus == null) {
            throw new OrderStatusConflictException(
                    "Order status is required"
            );
        }

        OrderStatus currentStatus = order.getStatus();

        if (currentStatus == requestedStatus) {
            throw new OrderStatusConflictException(
                    "Order already has status "
                            + requestedStatus
            );
        }

        if (requestedStatus == OrderStatus.CANCELLED) {
            return cancelOrder(order);
        }

        if (requestedStatus == OrderStatus.SHIPPED) {
            throw new OrderStatusConflictException(
                    "Use the shipment endpoint to ship an order"
            );
        }

        if (requestedStatus == OrderStatus.DELIVERED) {
            throw new OrderStatusConflictException(
                    "Use the shipment endpoint to mark an order as delivered"
            );
        }

        boolean validTransition =
                currentStatus == OrderStatus.PLACED
                        && requestedStatus == OrderStatus.PROCESSING;

        if (!validTransition) {
            throw new OrderStatusConflictException(
                    "Unsupported order status transition: "
                            + currentStatus
                            + " -> "
                            + requestedStatus
            );
        }

        Payment payment = paymentRepository
                .findByOrderId(orderId)
                .orElseThrow(
                        () -> new PaymentNotFoundException(orderId)
                );

        if (payment.getStatus() != PaymentStatus.PAID) {
            throw new OrderStatusConflictException(
                    "Order must be paid before fulfillment"
            );
        }

        order.setStatus(requestedStatus);

        Order savedOrder =
                orderRepository.save(order);

        return mapToResponse(savedOrder);
    }

    private AdminOrderResponse cancelOrder(
            Order order
    ) {
        OrderStatus currentStatus =
                order.getStatus();

        if (
                currentStatus != OrderStatus.PLACED
                        && currentStatus
                        != OrderStatus.PROCESSING
        ) {
            throw new OrderStatusConflictException(
                    "Order cannot be cancelled from status "
                            + currentStatus
            );
        }

        Payment payment = paymentRepository
                .findByOrderId(order.getId())
                .orElseThrow(
                        () -> new PaymentNotFoundException(
                                order.getId()
                        )
                );

        List<OrderItem> orderItems =
                new ArrayList<>(
                        orderItemRepository
                                .findByOrderIdOrderByIdAsc(
                                        order.getId()
                                )
                );

        orderItems.sort(
                Comparator.comparing(
                        item ->
                                item.getProduct().getId()
                )
        );

        for (OrderItem item : orderItems) {

            Long productId =
                    item.getProduct().getId();

            Product product = productRepository
                    .findByIdForUpdate(productId)
                    .orElseThrow(
                            () -> new IllegalStateException(
                                    "Product not found while cancelling order: "
                                            + productId
                            )
                    );

            int quantity =
                    item.getQuantity();

            int stockBefore =
                    product.getStock();

            int stockAfter =
                    Math.addExact(
                            stockBefore,
                            quantity
                    );

            product.setStock(stockAfter);

            productRepository.save(product);

            InventoryMovement movement =
                    new InventoryMovement();

            movement.setProduct(product);

            movement.setType(
                    InventoryMovementType.STOCK_IN
            );

            movement.setQuantity(quantity);
            movement.setStockBefore(stockBefore);
            movement.setStockAfter(stockAfter);

            inventoryMovementRepository.save(
                    movement
            );
        }

        if (
                payment.getStatus()
                        == PaymentStatus.PAID
        ) {
            payment.setStatus(
                    PaymentStatus.REFUNDED
            );

            paymentRepository.save(payment);
        }

        order.setStatus(
                OrderStatus.CANCELLED
        );

        Order savedOrder =
                orderRepository.save(order);

        return mapToResponse(savedOrder);
    }

    private AdminOrderResponse mapToResponse(
            Order order
    ) {
        User customer =
                order.getUser();

        List<OrderItemResponse> items =
                orderItemRepository
                        .findByOrderIdOrderByIdAsc(
                                order.getId()
                        )
                        .stream()
                        .map(
                                item ->
                                        new OrderItemResponse(
                                                item.getId(),
                                                item.getProduct()
                                                        .getId(),
                                                item.getProductName(),
                                                item.getSku(),
                                                item.getUnitPrice(),
                                                item.getQuantity(),
                                                item.getSubtotal()
                                        )
                        )
                        .toList();

        String paymentStatus =
                paymentRepository
                        .findByOrderId(
                                order.getId()
                        )
                        .map(
                                payment ->
                                        payment.getStatus()
                                                .name()
                        )
                        .orElse("PENDING");

        return new AdminOrderResponse(
                order.getId(),
                customer.getId(),
                customer.getFirstName()
                        + " "
                        + customer.getLastName(),
                customer.getEmail(),
                order.getStatus().name(),
                order.getTotalAmount(),
                items,
                order.getRecipientName(),
                order.getPhoneNumber(),
                order.getPostalCode(),
                order.getAddress(),
                order.getAddressDetail(),
                order.getDeliveryInstructions(),
                order.getCreatedAt(),
                order.getUpdatedAt(),
                paymentStatus
        );
    }
}
package com.springcommerce.backend.repository;

import com.springcommerce.backend.entity.Order;
import com.springcommerce.backend.entity.OrderStatus;

import jakarta.persistence.LockModeType;

import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Lock;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;

import java.util.List;
import java.util.Optional;

public interface OrderRepository
        extends JpaRepository<Order, Long> {

    List<Order> findByUserIdOrderByCreatedAtDesc(
            Long userId
    );

    Optional<Order> findByIdAndUserId(
            Long orderId,
            Long userId
    );

    @Lock(LockModeType.PESSIMISTIC_WRITE)
    @Query("""
            SELECT o
            FROM Order o
            WHERE o.id = :orderId
              AND o.user.id = :userId
            """)
    Optional<Order> findByIdAndUserIdForUpdate(
            @Param("orderId") Long orderId,
            @Param("userId") Long userId
    );

    @Lock(LockModeType.PESSIMISTIC_WRITE)
    @Query("""
            SELECT o
            FROM Order o
            WHERE o.id = :orderId
            """)
    Optional<Order> findByIdForUpdate(
            @Param("orderId") Long orderId
    );

    Page<Order> findAllByOrderByCreatedAtDesc(
            Pageable pageable
    );

    Page<Order> findByStatusOrderByCreatedAtDesc(
            OrderStatus status,
            Pageable pageable
    );
}

package com.springcommerce.backend.entity;

public enum OrderStatus {
    PLACED,
    PROCESSING,
    SHIPPED,
    DELIVERED,
    CANCELLED
}

package com.springcommerce.backend.entity;

public enum PaymentStatus {
    PENDING,
    PAID,
    FAILED,
    REFUNDED
}
